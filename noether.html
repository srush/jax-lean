Flux update

def fluxStep {R : Type} [Ring R] {n : ℕ} (p : Equiv.Perm (Fin n)) (flux : R → R) (x : Tensor R [n]) : Tensor R [n] := fun i => x i - (flux (x i) - flux (permute p x i))
